const CODIGO =
  "def two_sum(nums, target):\n" +
  "    count = len(nums)\n" +
  "    for i in range(len(nums)):\n" +
  "        for j in range(len(nums)):\n" +
  "            if nums[i] + nums[j] == target:\n" +
  "                return [i, j]";

/** Termos que entregariam a solução (o filtro de fala também barra, mas o modelo não deveria nem tentar). */
const SOLUCAO = /dicion[aá]rio|hash ?map|tabela hash|complemento|\bset\b|conjunto/i;

function sessao(id: string, semCodigo = false): SessionState {
  const s = getOrCreateSession(id);
  // Sem código: só o código inicial da questão (o candidato ainda está entendendo o problema).
  const codigo = semCodigo ? getProblem(s.config.problemId).public.starters.python ?? "" : CODIGO;
  appendEvent(s, { type: "session_started", sessionId: id, t: 0, configId: s.config.id, problemId: s.config.problemId, language: "python" });
  appendEvent(s, { type: "snapshot", sessionId: id, t: 0, code: codigo, version: 1, language: "python" });
  const q: QueuedQuestion = {
    id: "q-1",
    question: "Me fala um pouco desse loop interno. Por que o j começa onde começa?",
    followUpTopic: "o raciocínio por trás de onde o loop começa",
    topicLabel: "o loop interno",
    priority: 0.8,
    highlight: { startLine: 3, endLine: 3 },
    createdAt: 0,
    status: "blipped",
  };
  s.queue.push(q);
  s.blips.push({ id: "q-1", preview: "Tenho uma pergunta sobre seu código", priority: 0.8, createdAt: 0, highlight: q.highlight });
  return s;
}

type Caso = { nome: string; fala: string; responde: boolean; blip?: boolean; encerra?: boolean; semCodigo?: boolean; checar?: (r: string) => string | null; comentar?: string };

const CASOS: Caso[] = [
  { nome: "pensando em voz alta", fala: "então aqui eu percorro o array e pra cada número eu olho os outros", responde: false },
  { nome: "murmúrio", fala: "hmm deixa eu ver esse índice aqui", responde: false },
  { nome: "falando sozinho do erro", fala: "ué, deu errado no segundo teste, deixa eu olhar", responde: false },
  { nome: "dúvida do enunciado", fala: "posso assumir que sempre existe uma resposta?", responde: true },
  { nome: "pede a estrutura", fala: "qual estrutura de dados você usaria aqui?", responde: true, checar: (r) => (SOLUCAO.test(r) ? "entregou a solução" : null) },
  { nome: "pede validação", fala: "meu código tá certo?", responde: true, checar: (r) => (/\b(está|tá) certo\b|correto|isso mesmo/i.test(r) && !/\?/.test(r) ? "validou o código" : null) },
  { nome: "explica e pergunta", fala: "acho que fica O(n²) por causa dos dois loops, faz sentido pra você?", responde: true },
  { nome: "conta o plano antes de codar", fala: "vou começar pela força bruta, testando todos os pares, e depois vejo se dá pra melhorar", responde: true, semCodigo: true,
    checar: (r) => (SOLUCAO.test(r) ? "sugeriu a solução" : !/\?/.test(r) ? "não fez pergunta sobre o plano" : null) },
  { nome: "lendo o enunciado", fala: "hmm deixa eu ler de novo o enunciado aqui", responde: false, semCodigo: true },
  { nome: "pede para encerrar", fala: "acho que é isso, pode encerrar a entrevista", responde: false, encerra: true },
  { nome: "terminou a etapa", fala: "pronto, terminei a primeira versão e os testes passaram", responde: true, encerra: false },
  { nome: "pediu para olhar", fala: "Terminei a primeira versão, dá uma olhada no código.", responde: true, encerra: false, comentar: "count" },
  { nome: "resposta no balão", fala: "começa do zero porque eu quero olhar todos os pares", responde: true, blip: true },
];

async function main() {
  if (process.argv.includes("--offline")) {
    process.env.FEATHERLESS_API_KEY = "offline";
    process.env.SO_CASOS = String(CASOS.findIndex((caso) => caso.nome === "pediu para olhar"));
    globalThis.fetch = (async () => {
      const conteudo = { reason: "pergunta sobre decisão sem avaliar", safe: true, respond: true, concluded: true, reply: "O count registra o tamanho da lista. Como você pensou essa decisão?", highlight: { startLine: 2, endLine: 2 }, end: false, gaveHint: false, evidence: null, challenge: null, probeOutcome: null };
      return Response.json({ choices: [{ message: { content: JSON.stringify(conteudo) }, finish_reason: "stop" }], usage: {} });
    }) as typeof fetch;
  }
  const falhas: string[] = [];
  const tempos: number[] = [];
  const so = process.env.SO_CASOS?.split(",").map(Number);
  for (const [i, caso] of CASOS.entries()) {
    if (so && !so.includes(i)) continue;
    const s = sessao(`validar-ide-${Date.now()}-${i}`, caso.semCodigo);
    if (process.argv.includes("--offline")) s.codeHistory.push({ t: 0, version: 1, language: "python", code: CODIGO });
    const mensagens: ChatMessage[] = [];
    if (caso.blip) {
      appendEvent(s, { type: "blip_opened", sessionId: s.sessionId, t: 0, blipId: "q-1" });
      const abertura = await runInterviewerTurn(s, { blipId: "q-1", messages: [] });
      mensagens.push({ role: "agent", text: abertura.reply });
    }
    mensagens.push({ role: "candidate", text: caso.fala });
    const inicio = Date.now();
    const r = await runInterviewerTurn(s, { blipId: caso.blip ? "q-1" : undefined, messages: mensagens });
    const ms = Date.now() - inicio;
    tempos.push(ms);
    console.log(JSON.stringify({ caso: caso.nome, ms, silent: r.listening ?? false, end: r.end, despedida: !!r.despedida, reply: r.reply, commands: r.commands }));
    if (caso.encerra !== undefined && !!r.despedida !== caso.encerra) falhas.push(`${caso.nome}: ${caso.encerra ? "devia encerrar" : "não devia encerrar"}`);
    const respondeu = !r.listening && r.reply.trim() !== "";
    if (respondeu !== caso.responde) falhas.push(`${caso.nome}: ${caso.responde ? "devia responder" : "devia ficar quieto"}`);
    else if (respondeu) {
      const problema = caso.checar?.(r.reply);
      if (problema) falhas.push(`${caso.nome}: ${problema}`);
      if (caso.comentar) {
        if (!r.reply.includes(caso.comentar)) falhas.push(`${caso.nome}: comentário não citou ${caso.comentar}`);
        if (!r.reply.trim().endsWith("?")) falhas.push(`${caso.nome}: comentário não terminou em pergunta`);
        if (!r.commands.some((c) => c.type === "highlight_range")) falhas.push(`${caso.nome}: trecho comentado sem highlight`);
      }
      if (!r.end && !deixaABola(r.reply)) falhas.push(`${caso.nome}: não devolve a vez`);
    }
  }
  const ordenado = [...tempos].sort((a, b) => a - b);
  console.log(`p50 ${ordenado[Math.floor(ordenado.length / 2)]} ms, máx ${ordenado.at(-1)} ms`);
  assert.equal(falhas.length, 0, falhas.join("; "));
}
main()
  .catch((erro) => {
    console.error(erro instanceof Error ? erro.message : erro);
    process.exitCode = 1;
  })
  // A store da sessão deixa timers vivos: sem isso o processo não termina.
  .finally(() => setTimeout(() => process.exit(), 100));
